Sample Code for Clustering Students
from sklearn.cluster import KMeans
import numpy as np

# Sample student data: [hours_studied, past_score]
data = np.array([[5, 70], [10, 90], [2, 50], [8, 85], [4, 60]])

# Apply K-means with 2 clusters
kmeans = KMeans(n_clusters=2, random_state=42)
kmeans.fit(data)
labels = kmeans.labels_

print("Student Clusters:", labels)
